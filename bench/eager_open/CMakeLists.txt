# Per-open-index cost on Xapiand's open path (docs/perf/eager-open.md): a project of its own over
# prospect, whose strata is STRATA_SOURCE (this tree by default) and whose prospect is
# FETCHCONTENT_SOURCE_DIR_PROSPECT when set, else prospect's pin below.
cmake_minimum_required(VERSION 3.18)
project(eager_open LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
set(STRATA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "the strata tree prospect builds on")
set(FETCHCONTENT_SOURCE_DIR_STRATA "${STRATA_SOURCE}" CACHE PATH "" FORCE)
FetchContent_Declare(prospect GIT_REPOSITORY https://github.com/Kronuz/prospect.git GIT_TAG 0448b8774e7b2c485ac8e5afd25893068b377a8b)
FetchContent_GetProperties(prospect)
if(NOT prospect_POPULATED)
  FetchContent_Populate(prospect)
  add_subdirectory(${prospect_SOURCE_DIR} ${prospect_BINARY_DIR} EXCLUDE_FROM_ALL)
endif()
add_executable(eager_open eager_open.cc)
add_executable(eager_open_count eager_open.cc)
target_compile_definitions(eager_open_count PRIVATE IDLE_COUNT)
target_compile_options(eager_open_count PRIVATE -fno-omit-frame-pointer)
foreach(t eager_open eager_open_count)
  target_link_libraries(${t} PRIVATE prospect::prospect)
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
  target_link_libraries(eager_open_count PRIVATE execinfo)
endif()
